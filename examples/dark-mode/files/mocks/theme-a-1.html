Retry with backoff

JavaScript, edited yesterday

async function retry(fn, tries = 3) {
  for (let i = 0; i < tries; i++) {
    try { return await fn(); }
    catch (err) { await sleep(200 * 2 ** i); }
  }
  throw new Error("gave up");
}
